cmake_minimum_required(VERSION 3.20)
project(ManicGBLink C)
set(CMAKE_C_STANDARD 11)
set(SB ${CMAKE_CURRENT_SOURCE_DIR}/Vendor/SameBoy)
set(SOURCES gb sgb apu memory mbc timing display camera sm83_cpu joypad
            save_state random rumble)
list(TRANSFORM SOURCES PREPEND "${SB}/Core/")
list(TRANSFORM SOURCES APPEND ".c")
add_library(mgl STATIC Core/MGLCore.c ${SOURCES})
set_source_files_properties(${SOURCES} PROPERTIES COMPILE_DEFINITIONS GB_INTERNAL)
target_include_directories(mgl PUBLIC Core ${SB}/Core)
target_compile_definitions(mgl PUBLIC GB_DISABLE_TIMEKEEPING GB_DISABLE_REWIND
    GB_DISABLE_DEBUGGER GB_DISABLE_CHEATS GB_VERSION="1.0.3" GB_COPYRIGHT_YEAR="2026")
target_compile_definitions(mgl PRIVATE _GNU_SOURCE)
if(NOT WIN32)
    target_link_libraries(mgl PUBLIC m)
endif()
option(MGL_TESTING "Build legal synthetic core tests" ON)
if(MGL_TESTING)
    enable_testing()
    target_compile_definitions(mgl PUBLIC MGL_TEST)
    add_executable(mgl_tests Tests/core_test.c)
    target_link_libraries(mgl_tests PRIVATE mgl)
    add_test(NAME gb_serial_and_protocol COMMAND mgl_tests)
    add_executable(mgl_boot_tests Tests/boot_test.c)
    target_link_libraries(mgl_boot_tests PRIVATE mgl)
    if(UNIX)
        add_executable(mgl_network_tests Tests/network_test.c)
        target_link_libraries(mgl_network_tests PRIVATE mgl)
        add_test(NAME network_faults COMMAND mgl_network_tests)
        set_tests_properties(network_faults PROPERTIES TIMEOUT 20)
    endif()
endif()
