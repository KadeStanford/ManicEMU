cmake_minimum_required(VERSION 3.20)
project(ManicTrade C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GPSP ${CMAKE_CURRENT_SOURCE_DIR}/Vendor/gpSP)
set(COMMON ${GPSP}/libretro/libretro-common)
add_library(trade SHARED Core/TradeCore.c)
target_include_directories(trade PUBLIC Core)
target_link_libraries(trade PRIVATE pthread)
set(CORE_FILES main.c gba_memory.c savestate.c input.c sound.c cheats.c memmap.c
    serial.c gbp.c rfu.c serial_proto.c libretro/libretro.c gba_cc_lut.c cpu.cc video.cc)
set(COMMON_FILES compat/compat_posix_string.c compat/compat_strl.c compat/fopen_utf8.c
    encodings/encoding_utf.c file/file_path.c file/file_path_io.c streams/file_stream.c
    string/stdstring.c time/rtime.c vfs/vfs_implementation.c)
set(SOURCES)
foreach(f IN LISTS CORE_FILES)
    list(APPEND SOURCES ${GPSP}/${f})
endforeach()
foreach(f IN LISTS COMMON_FILES)
    list(APPEND SOURCES ${COMMON}/${f})
endforeach()
file(READ ${GPSP}/bios/open_gba_bios.bin BIOS_BYTES HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," BIOS_ARRAY "${BIOS_BYTES}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/open_bios.c "unsigned char open_gba_bios_rom[16384]={${BIOS_ARRAY}};\n")
add_library(gpsp SHARED ${SOURCES} ${CMAKE_CURRENT_BINARY_DIR}/open_bios.c)
target_include_directories(gpsp PRIVATE ${GPSP} ${GPSP}/libretro ${COMMON}/include Core)
target_compile_definitions(gpsp PUBLIC INLINE=inline __LIBRETRO__ HAVE_STDINT_H HAVE_INTTYPES_H)
target_compile_definitions(gpsp PRIVATE FRONTEND_SUPPORTS_RGB565 HAVE_POSIX_MEMALIGN HAVE_UNISTD_H HAVE_FCNTL_H HAVE_STRINGS_H HAVE_SYS_PARAM_H HAVE_SYS_STAT_H HAVE_SYS_TYPES_H HAVE_SYS_TIME_H HAVE_TIME_H HAVE_DIRENT_H HAVE_FTRUNCATE HAVE_64BIT_OFFSETS GIT_VERSION="fc4afeb-mtr4")
target_compile_options(gpsp PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>)
target_link_libraries(gpsp PRIVATE trade m)
if(APPLE)
    enable_language(OBJC)
    target_sources(trade PRIVATE iOS/ManicTrade.m)
    target_compile_options(trade PRIVATE $<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc;-Wall;-Wextra;-Werror;-Wno-unused-parameter>)
    target_link_libraries(trade PRIVATE "-framework Foundation" "-framework UIKit" "-framework MultipeerConnectivity" "-framework QuartzCore" "-framework CoreGraphics")
    set_target_properties(trade PROPERTIES OUTPUT_NAME ManicGBLink PREFIX "" SUFFIX "" INSTALL_NAME_DIR "@rpath/ManicGBLink.framework" BUILD_WITH_INSTALL_NAME_DIR TRUE)
    set_target_properties(gpsp PROPERTIES OUTPUT_NAME gpsp.libretro PREFIX "" SUFFIX "" INSTALL_NAME_DIR "@rpath/gpsp.libretro.framework" BUILD_WITH_INSTALL_NAME_DIR TRUE)
else()
    enable_testing()
    add_executable(trade_tests Tests/transport_test.c)
    target_link_libraries(trade_tests PRIVATE trade)
    add_test(NAME transport COMMAND trade_tests)
    add_executable(gba_tests Tests/gba_test.c)
    target_include_directories(gba_tests PRIVATE ${GPSP} ${GPSP}/libretro ${COMMON}/include)
    target_link_libraries(gba_tests PRIVATE gpsp trade)
    add_test(NAME gba_serial COMMAND gba_tests)
    add_executable(gba_peer Tests/gba_peer.c)
    target_include_directories(gba_peer PRIVATE ${GPSP} ${GPSP}/libretro ${COMMON}/include)
    target_link_libraries(gba_peer PRIVATE gpsp trade)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME gba_network COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/network_test.py $<TARGET_FILE:gba_peer>)
    add_test(NAME firered_transitions COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/transition_test.py $<TARGET_FILE:gba_peer>)
    add_test(NAME firered_room_exit COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/exit_test.py $<TARGET_FILE:gba_peer>)
    set_tests_properties(gba_network PROPERTIES TIMEOUT 90)
    set_tests_properties(firered_transitions PROPERTIES TIMEOUT 120)
    set_tests_properties(firered_room_exit PROPERTIES TIMEOUT 120)
    add_executable(save_migrate Tests/save_migrate.c)
    target_include_directories(save_migrate PRIVATE ${COMMON}/include)
    target_link_libraries(save_migrate PRIVATE dl)
endif()
