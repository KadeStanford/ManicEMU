target_link_libraries(ds_address_map PRIVATE ds_protocol)

target_link_libraries(ds_gen5_identity PRIVATE ds_protocol)
add_executable(ds_native_replies Tests/native_replies.cpp)
target_link_libraries(ds_native_replies PRIVATE ds_protocol)
add_executable(ds_native_radio Tests/native_radio.cpp)
target_link_libraries(ds_native_radio PRIVATE ds_protocol)

if(MSVC)
 target_compile_options(ds_protocol PRIVATE /W4 /WX)

 target_compile_options(ds_protocol PRIVATE -Wall -Wextra -Werror)
 target_compile_options(ds_peer PRIVATE -Wall -Wextra -Werror)
 target_compile_options(ds_address_map PRIVATE -Wall -Wextra -Werror)
 target_compile_options(ds_batch_wait PRIVATE -Wall -Wextra -Werror)

enable_testing()

add_test(NAME native_empty_reply_and_payload_receive COMMAND ds_native_replies)
add_test(NAME bounded_native_rf_datagrams COMMAND ds_native_radio)
add_test(NAME room_native_rf_controls_and_repeated_sessions COMMAND ds_room_radio)
add_test(NAME identity_preserving_address_map COMMAND ds_address_map)
add_test(NAME bounded_batch_and_wall_clock_receive COMMAND ds_batch_wait)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MANIC_DS_NATIVE_SOURCE "" CACHE PATH "Pinned, fully patched public melonDS DS checkout for actual native boundary regression")
if(MANIC_DS_NATIVE_SOURCE)
 execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/generate_native_wait.py
  --source ${MANIC_DS_NATIVE_SOURCE} --output ${CMAKE_CURRENT_BINARY_DIR}/native_wait_functions.inc
  RESULT_VARIABLE native_wait_generated)
 if(NOT native_wait_generated EQUAL 0)
  message(FATAL_ERROR "Actual native receive regression source guard failed")
 endif()
 add_executable(ds_native_wait Tests/native_wait.cpp)
 target_link_libraries(ds_native_wait PRIVATE ds_protocol)
 target_compile_definitions(ds_native_wait PRIVATE MANIC_DS_WAIT_FUNCTIONS="${CMAKE_CURRENT_BINARY_DIR}/native_wait_functions.inc")
 if(MSVC)
  target_compile_options(ds_native_wait PRIVATE /W4 /WX)
 else()
  target_compile_options(ds_native_wait PRIVATE -Wall -Wextra -Werror)
 endif()
 add_test(NAME actual_native_final_wait_boundary COMMAND ds_native_wait)
 add_executable(ds_native_send Tests/native_send.cpp)
 target_link_libraries(ds_native_send PRIVATE ds_protocol)
 target_compile_definitions(ds_native_send PRIVATE MANIC_DS_SEND_FUNCTION="${CMAKE_CURRENT_BINARY_DIR}/native_send_function.inc")
 if(MSVC)
  target_compile_options(ds_native_send PRIVATE /W4 /WX)
 else()
  target_compile_options(ds_native_send PRIVATE -Wall -Wextra -Werror)
 endif()
 add_test(NAME actual_native_diagnostic_aid_preserves_bytes COMMAND ds_native_send)
endif()
