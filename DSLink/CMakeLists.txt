cmake_minimum_required(VERSION 3.20)
project(ManicDSProtocol LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(ds_protocol Core/Protocol.cpp)
target_include_directories(ds_protocol PUBLIC Core)
add_executable(ds_peer Tests/peer.cpp)
target_link_libraries(ds_peer PRIVATE ds_protocol)
add_executable(ds_address_map Tests/address_map.cpp)
target_link_libraries(ds_address_map PRIVATE ds_protocol)
add_executable(ds_batch_wait Tests/batch_wait.cpp)
target_link_libraries(ds_batch_wait PRIVATE ds_protocol)
if(MSVC)
 target_compile_options(ds_protocol PRIVATE /W4 /WX)
 target_compile_options(ds_peer PRIVATE /W4 /WX)
 target_compile_options(ds_address_map PRIVATE /W4 /WX)
 target_compile_options(ds_batch_wait PRIVATE /W4 /WX)
else()
 target_compile_options(ds_protocol PRIVATE -Wall -Wextra -Werror)
 target_compile_options(ds_peer PRIVATE -Wall -Wextra -Werror)
 target_compile_options(ds_address_map PRIVATE -Wall -Wextra -Werror)
 target_compile_options(ds_batch_wait PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME identity_preserving_address_map COMMAND ds_address_map)
add_test(NAME bounded_batch_and_wall_clock_receive COMMAND ds_batch_wait)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME independent_protocol_processes COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/protocol_test.py $<TARGET_FILE:ds_peer> ${CMAKE_CURRENT_BINARY_DIR}/protocol.json)
set_tests_properties(independent_protocol_processes PROPERTIES TIMEOUT 120)
